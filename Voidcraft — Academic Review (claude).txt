Three facts make an exact, scale-invariant version possible. A cube whose whole column down to row 0 is solid is attached to bedrock by construction (one cumulative AND). Only the leftover overhang cubes can be loose. Those can be labelled inside their own bounding box. The result is identical to labelling the whole box.```python# fixes/islands.pyimport mathimport numpy as npfrom scipy import ndimage_FACE = ndimage.generate_binary_structure(3, 1)   # 6-connectivity, same as ndimage.label's defaultdef floating_islands(E, world, center, radius, margin=8.0, up_margin=3.0):    """Groups of solid ground cubes cut off from bedrock and from the scan boundary.    Returns a list of (n, 3) int arrays of cube indices. Same answer as World.settle_islands where    that function's box is not shrunk. margin is honoured at every cube detail."""    cell = E.CELL    c0 = np.floor(np.asarray(center, np.float64) / cell).astype(np.int64)    h = int(math.ceil((radius + margin) / cell)) + 2    up = int(math.ceil((radius + up_margin) / cell)) + 2    lo = np.array([c0[0] - h, 0, c0[2] - h])                      # always reaches bedrock row 0    hi = np.array([c0[0] + h, min(E.NH - 1, c0[1] + up), c0[2] + h])    if hi[1] < lo[1]:        return []    box = world.cells_box(lo, hi)    grounded = np.logical_and.accumulate(box, axis=1)            # solid run from row 0 upward    rest = box & ~grounded                                        # overhangs and anything on an air gap    if not rest.any():        return []    nz = [np.nonzero(rest.any(axis=tuple(a for a in range(3) if a != k)))[0] for k in range(3)]    b0 = np.array([max(int(v[0]) - 1, 0) for v in nz])            # bounding box of the overhang,    b1 = np.array([min(int(v[-1]) + 2, s) for v, s in zip(nz, box.shape)])   # padded by one cube    sl = tuple(slice(int(a), int(b)) for a, b in zip(b0, b1))    r, g = rest[sl], grounded[sl]    lab, n = ndimage.label(r, structure=_FACE)    keep = np.zeros(n + 1, bool)    keep[0] = True    for k in range(3):                                            # touches the scan boundary: may continue beyond it        if k != 1 and b0[k] == 0:            keep[np.unique(lab.take(0, axis=k))] = True        if b1[k] == box.shape[k]:            keep[np.unique(lab.take(-1, axis=k))] = True    touch = ndimage.binary_dilation(g, structure=_FACE) & r      # face-adjacent to a grounded cube    keep[np.unique(lab[touch])] = True    floating = r & ~keep[lab]    if not floating.any():        return []    idx = np.argwhere(floating)    labs = lab[floating]    order = np.argsort(labs, kind='stable')    idx, labs = idx[order], labs[order]    cuts = np.nonzero(np.diff(labs))[0] + 1    return [grp + b0 + lo for grp in np.split(idx, cuts)]```Verification. This is the first piece of the event bus in section 3.```python# World.__init__self.listeners = []     # callables fn(lo, hi): world-space corners of every box of ground cubes just removed# World._changed_chunks, right after `self.edit_ver += 1`if self.listeners:    box = (np.asarray(lo, np.float64) * CELL, (np.asarray(hi, np.float64) + 1.0) * CELL)    for fn in self.listeners:        fn(*box)# DirtCubesdef wake_box(self, lo, hi, pad=0.05, column_up=3.0):    """Wake cubes touching the box, and piles standing on its footprint up to `column_up` above it."""    if self.n == 0:        return    p, h = self.pos, self.size[:, None] * 0.5    lo, hi = np.asarray(lo, np.float64), np.asarray(hi, np.float64)    foot = ((p[:, [0, 2]] + h >= lo[[0, 2]] - pad) & (p[:, [0, 2]] - h <= hi[[0, 2]] + pad)).all(axis=1)    hit = foot & (p[:, 1] + h[:, 0] >= lo[1] - pad) & (p[:, 1] - h[:, 0] <= hi[1] + column_up)    self.awake[hit] = True    self.still[hit] = 0.0# StructureManagerdef wake_bodies_in_box(self, lo, hi, pad=0.1):    lo = np.asarray(lo, np.float64) - pad    hi = np.asarray(hi, np.float64) + pad    for b in self.bodies:        if b.asleep:            blo, bhi = b.aabb()            if (blo <= hi).all() and (bhi >= lo).all():                b.wake()# in run(), once, after World() and the singletons existworld.listeners.append(lambda lo, hi: (BODIES.wake_box(lo, hi), STRUCTS.wake_bodies_in_box(lo, hi)))```The wake range is conservative by design (a whole removed box, not the exact cube set), and `column_up = 3` blocks mirrors the existing `on_terrain_changed(..., column_up=radius * 3.0)` convention. Note `_restore_stone` writes `d.arr` directly; route it through the same notification (or make bedrock a property of `World.dig` itself) so the rule stops living in game code.4.4 Rigid-body ground early-out is dead code (`RigidBody._ground_top`)Defect. `_contacts` skips terrain tests for surface points above `self._top + 0.4`, and `step` sets `_top` from `_ground_top()`, which returns the constant `TERRAIN_MAX_H` (48 blocks). No point is ever above that, so a slab falling through open air still samples the ground seven times per point (six neighbours plus the cell itself) for 1,500 to 3,000 points, three substeps per frame. A profile of 8 slabs showed `_contacts` at about 73% of the step, with `_cell_contact` and `_sample_many` underneath it.Fix. Ground is only ever removed, so a chunk's original maximum height is a valid ceiling forever. Cache it on `ChunkData` and take the maximum over the chunks under the body's box:```python# ChunkData__slots__ = ('top', 'arr', 'hmax')def __init__(self, top):    self.top = top    self.arr = None    self.hmax = float(top.max()) * CELL     # digging only lowers the ground, so this stays a valid ceiling# RigidBodydef _ground_top(self, world):    lo, hi = self.aabb()    top = 0.0    for kx in range(math.floor(lo[0] / CHUNK_SIZE), math.floor(hi[0] / CHUNK_SIZE) + 1):        for kz in range(math.floor(lo[2] / CHUNK_SIZE), math.floor(hi[2] / CHUNK_SIZE) + 1):            d = world.dens.get((kx, kz))            if d is not None:                top = max(top, d.hmax)    return top```Verification.
